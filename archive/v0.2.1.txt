// constant timers
#define RED_TIME     15000 // RED time in milliseconds
#define YELLOW_TIME  4000 // YELLOW time in milliseconds
#define GREEN_TIME   20000 // GREEN time in milliseconds
#define RAPID_BUZZER 150
#define NORM_BUZZER  500

// global timestamps (for millis)
unsigned long greenStartedAt;
unsigned long yellowStartedAt;
unsigned long redStartedAt;
unsigned long buzzerStartedAt;
bool isBuzzing = false;

// # function declarations
void phaseRed();
void phaseYellow();
void phaseGreen();
void buzzerPing(int type);

// # setup code runs once after boot
void setup() {
  Serial.begin(9600);
  pinMode(PIN_RED, OUTPUT); // leds output light
  pinMode(PIN_YELLOW, OUTPUT);
  pinMode(PIN_GREEN, OUTPUT);

  pinMode(PIN_BUZZER, OUTPUT); // buzzers output sound
  // pinMODE(PIN_BUTTON, INPUT_PULLUP); // buttons input presses

}

// # loop function runs after setup indefinitely (thousands of times a second)
void loop() {
  int currentPhase; // tracks current phase (0=red,1=yellow,2=green)

  // signal loop based on US traffic signals
  if (millis() - redStartedAt >= RED_TIME) {
    phaseGreen();
    greenStartedAt = millis();
  }

  if (millis() - greenStartedAt >= GREEN_TIME) {
    phaseYellow();
    yellowStartedAt = millis();
  }

  if (millis() - yellowStartedAt >= YELLOW_TIME) {
    phaseRed();
    redStartedAt = millis;
  }

}
