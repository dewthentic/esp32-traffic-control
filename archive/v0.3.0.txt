// # constants
// traffic light pins
#define PIN_RED    18 // The ESP32 pin GPIO18 connected to R pin of traffic light module
#define PIN_YELLOW 21 // The ESP32 pin GPIO21 connected to Y pin of traffic light module
#define PIN_GREEN  22 // The ESP32 pin GPIO22 connected to G pin of traffic light module

// buzzer pin (to-do: add button pin)
#define PIN_BUZZER 19

// constant timers
#define RED_TIME     15000 // RED time in milliseconds
#define YELLOW_TIME  4000 // YELLOW time in milliseconds
#define GREEN_TIME   20000 // GREEN time in milliseconds
#define RAPID_BUZZER 150
#define NORM_BUZZER  500

// global timestamps (for millis)
unsigned long greenStartedAt;
unsigned long yellowStartedAt;
unsigned long redStartedAt;
// unsigned long buzzerStartedAt;
// bool isBuzzing = false;

// # function declarations
void phaseRed();
void phaseYellow();
void phaseGreen();
void buzzerPing(int type);

// # setup code runs once after boot
void setup() {
  Serial.begin(9600);
  pinMode(PIN_RED, OUTPUT); // leds output light
  pinMode(PIN_YELLOW, OUTPUT);
  pinMode(PIN_GREEN, OUTPUT);

  pinMode(PIN_BUZZER, OUTPUT); // buzzers output sound
  // pinMODE(PIN_BUTTON, INPUT_PULLUP); // buttons input presses

  phaseRed(); // start the program as a red light
  redStartedAt = millis();

}

// # loop function runs after setup indefinitely (thousands of times a second)
void loop() {
  int currentPhase; // tracks current phase (0=red,1=yellow,2=green)
  
  // using millis() requires a mindset change from telling a program to stay put (delay()) to being a security guard waiting for the right time to do something (millis())
  // i.e. state machines

  // signal loop based on US traffic signals
  // switch statement keeps this clean, only worry about one signal at a time
  switch (currentPhase) {
    case 0:
      if (millis() - redStartedAt >= RED_TIME) {
        // it's time to switch to another state, switch to another state and check the time there.
        phaseGreen();
        currentPhase = 2;
        greenStartedAt = millis();
      }
      // reaches the end = it's not time yet. repeat the loop and stay in current phase.

    case 1:
      if (millis() - yellowStartedAt >= YELLOW_TIME) {
        phaseRed();
        currentPhase = 0;
        redStartedAt = millis();
      }

    case 2:
      if (millis() - greenStartedAt >= GREEN_TIME) {
        phaseYellow();
        currentPhase = 1;
        yellowStartedAt = millis();
      }
    
    default:
    // if the signal is a primary signal, flash yellow every second
    // otherwise, the signal is an arterial signal, flash red every second
  }

}
  
// # function definitions

// buzzer ping type used for pedestrian sound cues
void buzzerPing(int type) {
  int ticksPerSecond = 10;
  switch (type) {
    case 0: // one beep per second (timer counting down)
      digitalWrite(PIN_BUZZER, HIGH);
      delay(125);
      break;
    case 1: // burst (walk sign active, 8-10 beeps per second)
      while (ticksPerSecond > 0) {
        digitalWrite(PIN_BUZZER, LOW);
        delay(125);
        digitalWrite(PIN_BUZZER, HIGH);
        ticksPerSecond--;
      }
      break;
    case 2: // end (timer terminated)
      digitalWrite(PIN_BUZZER, HIGH);
      delay(250);
  }
  digitalWrite(PIN_BUZZER, LOW); // turn off buzzer
}

// red light on
void phaseRed() {
  digitalWrite(PIN_RED, HIGH);   // turn on
  digitalWrite(PIN_YELLOW, LOW); // turn off
  digitalWrite(PIN_GREEN, LOW);  // turn off
  
  // sunsetted delay() for millis(), it stops the whole program which is NOT good!
  // delay(RED_TIME); // keep red light on during a period of time
}

// yellow light transition to red
void phaseYellow() {
  digitalWrite(PIN_RED, LOW);    // turn off red
  digitalWrite(PIN_YELLOW, HIGH); // turn on yellow
  digitalWrite(PIN_GREEN, LOW); // turn off green
  
  // delay(YELLOW_TIME); // keep yellow light on during a period of time
}

// green light on
void phaseGreen() {
  digitalWrite(PIN_RED, LOW);    // turn off red
  digitalWrite(PIN_YELLOW, LOW); // turn off yellow
  digitalWrite(PIN_GREEN, HIGH); // turn on green
  
  // delay(GREEN_TIME); // keep green light on during a period of time
}
