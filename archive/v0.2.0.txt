// # function definitions

// buzzer ping type used for pedestrian sound cues
void buzzerPing(int type) {
  int ticksPerSecond = 10;
  switch (type) {
    case 0: // one beep per second (timer counting down)
      digitalWrite(PIN_BUZZER, HIGH);
      delay(125);
      break;
    case 1: // burst (walk sign active, 8-10 beeps per second)
      while (ticksPerSecond > 0) {
        digitalWrite(PIN_BUZZER, LOW);
        delay(125);
        digitalWrite(PIN_BUZZER, HIGH);
        ticksPerSecond--;
      }
      break;
    case 2: // end (timer terminated)
      digitalWrite(PIN_BUZZER, HIGH);
      delay(250);
  }
  digitalWrite(PIN_BUZZER, LOW); // turn off buzzer
}

// red light on
void phaseRed() {
  digitalWrite(PIN_RED, HIGH);   // turn on
  digitalWrite(PIN_YELLOW, LOW); // turn off
  digitalWrite(PIN_GREEN, LOW);  // turn off
  delay(RED_TIME); // keep red light on during a period of time
}

// yellow light transition to red
void phaseYellow() {
  digitalWrite(PIN_BUZZER, LOW); // turn off buzzer (from green)

  digitalWrite(PIN_RED, LOW);    // turn off red
  digitalWrite(PIN_YELLOW, HIGH); // turn on yellow
  digitalWrite(PIN_GREEN, LOW); // turn off green
  delay(YELLOW_TIME); // keep yellow light on during a period of time
}

// green light on
void phaseGreen() {
  digitalWrite(PIN_RED, LOW);    // turn off red
  digitalWrite(PIN_YELLOW, LOW); // turn off yellow
  digitalWrite(PIN_GREEN, HIGH); // turn on green

  buzzerPing(1); // sound buzzer (burst 2x)
  delay(1000);
  buzzerPing(1);
  
  delay(GREEN_TIME); // keep green light on during a period of time
}
